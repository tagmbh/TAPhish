# Phase 3.45 — Quick-Start Wizard completion + engagement linkage

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Finish the Quick-Start Wizard (Steps 4–7), link engagements to campaigns, surface captures + 2FA codes in every report and dashboard, and make scanner traffic filterable.

**Architecture:** Hybrid reuse-first base + foundation helpers. Bolt new wizard steps onto the existing `QuickStart.php`, wrap existing dispatcher functions where possible, but every new risky helper (DKIM keygen, preflight gates, recipient import, launch orchestrator) ships as its own pure-helper module with injectable seams + PHPUnit coverage. Status transitions use compare-and-swap (`UPDATE ... WHERE status = $expected`) so a double-click can't double-launch.

**Tech Stack:** PHP 8.x + mysqli, jQuery + Bootstrap (existing), PHPUnit 10.5 (pure-helper tier), AES-256-GCM at-rest envelope (Phase 3.27/3.38 pattern), GitHub Actions FTPS deploy.

```php
// spear/manager/scanner_detect.php — append before the closing PHP tag
if (!function_exists('taphish_scanner_ensure_schema')) {
    function taphish_scanner_ensure_schema(\mysqli $conn): void
    {
        $columns = [
            ['tb_data_mailcamp_live',      'is_scanner',     "TINYINT(1) NOT NULL DEFAULT 0"],
            ['tb_data_mailcamp_live',      'scanner_reason', "VARCHAR(160) NULL"],
            ['tb_data_quick_tracker_live', 'is_scanner',     "TINYINT(1) NOT NULL DEFAULT 0"],
            ['tb_data_webform_submit',     'is_scanner',     "TINYINT(1) NOT NULL DEFAULT 0"],
        ];
        foreach ($columns as [$table, $col, $type]) {
            $stmt = $conn->prepare(
                "SELECT COUNT(*) FROM information_schema.COLUMNS
                 WHERE TABLE_SCHEMA = DATABASE() AND TABLE_NAME = ? AND COLUMN_NAME = ?"
            );
            if ($stmt === false) continue;
            $stmt->bind_param('ss', $table, $col);
            $stmt->execute();
            $present = (int) $stmt->get_result()->fetch_row()[0];
            $stmt->close();
            if ($present === 0) {
                @$conn->query("ALTER TABLE `$table` ADD COLUMN `$col` $type");
            }
        }
    }
}

if (!function_exists('taphish_should_filter_scanner_in_kpis')) {
    function taphish_should_filter_scanner_in_kpis(array $opts): bool
    {
        return empty($opts['include_scanner_hits']);
    }
}
```

```php
    {
        if (!taphish_engagement_validate_transition($from, $to)) return false;

        $changed = $stmt->affected_rows > 0;
        $stmt->close();
        return $ok && $changed;
    }
```

```php
        $pkey_new     = $pkey_new     ?? 'openssl_pkey_new';
        $pkey_export  = $pkey_export  ?? 'openssl_pkey_export';
        $pkey_details = $pkey_details ?? 'openssl_pkey_get_details';
        $key = $pkey_new(['private_key_bits' => 2048, 'private_key_type' => OPENSSL_KEYTYPE_RSA]);
        if (!$key) return ['ok' => false, 'error' => 'openssl_pkey_new unavailable or disabled'];
        $private = '';
        $pkey_export($key, $private);
        $details = $pkey_details($key);
        if (!$details || empty($details['key'])) return ['ok' => false, 'error' => 'cannot extract public key'];
        $pub_b64 = preg_replace('/-----.*?-----|\s+/s', '', $details['key']);
        return ['ok' => true, 'private_key_pem' => $private, 'public_key_b64' => $pub_b64];
    }
}

    {
        return 'v=DKIM1; k=rsa; p=' . $pub_b64;
    }
}
```

```php
if (!function_exists('taphish_engagement_launch')) {
    function taphish_engagement_launch(
        \mysqli $conn,
        array $ctx,
        ?callable $preflight_fn = null,
        ?callable $logIt_fn = null
    ): array {
        $logIt_fn = $logIt_fn ?? 'logIt';
        // 1. Validate
        $v = taphish_engagement_launch_validate($ctx);
        if (!$v['ok']) return $v;
        // 2. Preflight
        $preflight = $preflight_fn ? $preflight_fn($conn, $ctx) : taphish_preflight_run_all($conn, $ctx);
        if (!$preflight['ok']) return ['ok' => false, 'gates' => $preflight['gates']];
        // 3. CAS status transition (draft → live)
        $transitioned = taphish_engagement_transition_status($conn, $ctx['engagement_id'], 'draft', 'live');
        if (!$transitioned) {
            return ['ok' => false, 'errors' => ['Engagement already launched or cancelled']];
        }
        // 4. Build + insert campaign row
        $row = taphish_campaign_row_build($ctx);
        $stmt = $conn->prepare("INSERT INTO tb_core_mailcamp_list (campaign_id, campaign_name, ...) VALUES (?, ?, ...)");
        // ... bind + execute ...
        if (!$ok) {
            // Rollback the status transition
            taphish_engagement_transition_status($conn, $ctx['engagement_id'], 'live', 'draft');
            return ['ok' => false, 'errors' => ['Campaign insert failed']];
        }
        $logIt_fn('Engagement launched: ' . $ctx['slug'] . ' → campaign ' . $row['campaign_id']);
        return ['ok' => true, 'campaign_id' => $row['campaign_id'], 'engagement_id' => $ctx['engagement_id']];
    }
}
```
